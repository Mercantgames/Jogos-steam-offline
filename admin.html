async function uploadEPublicar() { const fileInput = document.getElementById('arquivo'); const nome = document.getElementById('nome').value; const preco = document.getElementById('preco').value; if (!fileInput.files[0] || !nome || !preco) return alert("Preencha tudo!"); const URL_BASE = "https://grogiwbccvnbyjgxvafw.supabase.co"; const KEY = "sb_publishable_6XLbmbjPmopxhOm2-_iwRw_6o0mQg14"; const file = fileInput.files[0]; const fileName = `${Date.now()}_${file.name}`; const filePath = `${URL_BASE}/storage/v1/object/public/capas/${fileName}`; try { // 1. Sobe a imagem const uploadRes = await fetch(`${URL_BASE}/storage/v1/object/capas/${fileName}`, { method: 'POST', headers: { 'apikey': KEY, 'Authorization': `Bearer ${KEY}` }, body: file }); if (!uploadRes.ok) throw new Error("Erro no Storage"); // 2. TENTA SALVAR (Testando os nomes das colunas) const res = await fetch(`${URL_BASE}/rest/v1/jogos`, { method: 'POST', headers: { 'apikey': KEY, 'Authorization': `Bearer ${KEY}`, 'Content-Type': 'application/json', 'Prefer': 'return=minimal' }, body: JSON.stringify({ nome: nome, preco: parseFloat(preco), url_da_imagem: filePath, // Nome que vimos na sua imagem 420 categoria: "lancamentos" }) }); if (res.ok) { alert("🚀 SUCESSO! Jogo na loja!"); window.location.reload(); } else { const erroDetalhado = await res.json(); console.error(erroDetalhado); alert("❌ Erro do Banco: " + erroDetalhado.message); } } catch (e) { alert("❌ Erro: " + e.message); } }
